{{/* Photo gallery shortcode. Local addition (the theme's own `gallery` shortcode was evaluated and rejected: it needs a leaf bundle, fixes thumbnails at 190px tall, and fails the build on an empty album). Uses only core Hugo image processing, so it survives theme bumps. Usage (in a content file): {{< photo-gallery >}} {{< photo-gallery columns="2" >}} {{< photo-gallery dir="media/gallery/fieldwork" >}} Photos live in `assets/media/gallery/` (or the `dir` given above). They are ordered by filename, so prefix them `10-`, `20-`, ... to control the order. The words under each photo come from `data/gallery.yaml`, keyed by filename. Either a bare string, or a heading plus body text: 10-lab.jpg: "Aligning the RA-SHG optical path" 20-thz.jpg: title: "Terahertz generation" text: "Intense terahertz fields are generated by optical rectification of near-infrared pulses in a LiNbO3 prism." Markdown works in both fields. Photos with no entry simply render without text. */}} {{ $dir := .Get "dir" | default "media/gallery" }} {{ $columns := .Get "columns" | default "1" }} {{ $pattern := printf "%s/*.{jpg,jpeg,png,JPG,JPEG,PNG,webp,WEBP}" $dir }} {{ $images := resources.Match $pattern }} {{ $captions := site.Data.gallery }} {{/* Narrower columns need smaller files. 2000px suits a full-width single column in the widened grid; at two columns each slot is roughly half that. The floor keeps a 3-column layout still sharp on HiDPI screens. */}} {{ $column_count := int $columns }} {{ if lt $column_count 1 }}{{ $column_count = 1 }}{{ end }} {{ $display_width := int (math.Max 600 (div 2000 $column_count)) }} {{/* Every photo is cropped to one aspect ratio so the grid rows line up. The default 3:2 matches the landscape shots straight off a DSLR. `anchor="Top"` keeps the top of the frame and trims the bottom — change it to Center or Smart if a subject sits low in the frame. */}} {{ $ratio := .Get "ratio" | default "3:2" }} {{ $anchor := .Get "anchor" | default "Top" }} {{ $ratio_parts := split $ratio ":" }} {{ $display_height := div (mul $display_width (int (index $ratio_parts 1))) (int (index $ratio_parts 0)) }} {{ if $images }}
{{ range sort $images "Name" }} {{ $name := path.Base .Name }} {{/* An entry is either a bare string or a map with `title` / `text`, plus an optional `anchor` for photos the shared crop would otherwise spoil (a subject low in a portrait frame, say). */}} {{ $title := "" }} {{ $text := "" }} {{ $item_anchor := $anchor }} {{ with $captions }} {{ with index . $name }} {{ if reflect.IsMap . }} {{ $title = .title | default "" }} {{ $text = .text | default "" }} {{ $item_anchor = .anchor | default $anchor }} {{ else }} {{ $text = . }} {{ end }} {{ end }} {{ end }} {{/* The lightbox and the `alt` attribute need plain text, not the rendered Markdown the visible caption uses. */}} {{ $plain := chomp (plainify (markdownify (trim (printf "%s %s" $title $text) " "))) }} {{/* Grid thumbnail: cropped to the shared ratio so every row is even. The lightbox keeps the *uncropped* frame, so clicking through still shows the whole photo. Note the explicit if/else on the lightbox copy: `cond` evaluates both branches, which would upscale smaller photos. */}} {{ $display := .Fill (printf "%dx%d %s" $display_width $display_height $item_anchor) }} {{ $full := . }} {{ if gt .Width 2400 }} {{ $full = .Resize "2400x" }} {{ end }} {{ end }}
{{ else }}

No photos yet. Drop image files into assets/{{ $dir }}/ and they will appear here, ordered by filename.
